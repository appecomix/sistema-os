CONSULTA DE OS // ------------------------------------------------------------------ if (data.action === "buscarOS") { var docOS = SpreadsheetApp.getActiveSpreadsheet(); var sheetOS = docOS.getSheetByName("Dados"); var ultLinha = sheetOS.getLastRow(); if (ultLinha < 2) { return ContentService.createTextOutput(JSON.stringify({ "result": "success", "lista": [] })) .setMimeType(ContentService.MimeType.JSON); } var dados = sheetOS.getRange(2, 1, ultLinha - 1, 37).getValues(); var resultados = []; for (var r = 0; r < dados.length; r++) { var row = dados[r]; var numOS = String(row[1] || ""); var dtInicial = String(row[6] || ""); var tecnico = String(row[31] || ""); var equipe = String(row[30] || ""); if (!data.isSupervisor && tecnico.toLowerCase() !== String(data.usuarioLogado).toLowerCase()) { continue; } var matchOS = !data.usarOS || (data.filtroOS && numOS.toLowerCase().includes(String(data.filtroOS).toLowerCase())); var matchData = !data.usarData || (data.filtroData && dtInicial.includes(String(data.filtroData))); var matchTecnico = !data.usarTecnico || (data.filtroTecnico && tecnico.toLowerCase().includes(String(data.filtroTecnico).toLowerCase())); var matchEquipe = !data.usarEquipe || (data.filtroEquipe && equipe.toLowerCase().includes(String(data.filtroEquipe).toLowerCase())); if (matchOS && matchData && matchTecnico && matchEquipe) { resultados.push({ numero_os: numOS, controladora: row[2], tipo_servico: row[3] === "X" ? "Corretivo" : (row[4] === "X" ? "Preventivo" : "Implantação"), data_inicial: dtInicial, hora_inicial: row[7], data_final: row[8], hora_final: row[9], descricao_servico: row[10], tensao_alternada: row[11], tensao_75vdc: row[12], tensao_12vdc: row[13], tensao_24vdc: row[14], aterramento: row[15], cabeamento: row[16], fusivel: row[17], involucro: row[18], gabinetes: row[19], placas_cpu_potencia: row[20], padronizada: row[21], blocos_semaforicos: row[22], colunas: row[23], bracos: row[24], status_servico: row[25] === "X" ? "Concluído" : (row[26] === "X" ? "Pendente" : ""), pendencia: row[27], material: row[28], observacao: row[29], equipe: equipe, tecnico: tecnico, supervisor: row[32], codigo_servico: row[33], nobreak: row[34], status_controlador: row[35] === "X" ? "Operacional" : (row[36] === "X" ? "Inoperante" : "") }); } } return ContentService.createTextOutput(JSON.stringify({ "result": "success", "lista": resultados })) .setMimeType(ContentService.MimeType.JSON); } // ------------------------------------------------------------------ // 5. CARREGAR AUXILIARES // ------------------------------------------------------------------ if (data.action === "getControladoras") { var ID_PLANILHA_CTRL = "1PEqr3WGnwxPeNB3R10PLB5JeVM54l0ek7y9ulBrj7_A"; var docCtrl = SpreadsheetApp.openById(ID_PLANILHA_CTRL); var sheetCtrl = docCtrl.getSheets()[0]; var dadosCtrl = sheetCtrl.getRange(2, 1, sheetCtrl.getLastRow() - 1, 2).getValues(); var listaControladoras = []; for (var j = 0; j < dadosCtrl.length; j++) { var codigo = String(dadosCtrl[j][0]).trim(); var endereco = String(dadosCtrl[j][1]).trim(); if (endereco !== "") { listaControladoras.push(codigo !== "" ? codigo + " - " + endereco : endereco); } } return ContentService.createTextOutput(JSON.stringify({ "result": "success", "controladoras": listaControladoras })).setMimeType(ContentService.MimeType.JSON); } if (data.action === "getCodigosServico") { var ID_PLANILHA_CODIGOS = "1k87SyixjdvwdN6QLjvQ4TVxZZsykTVG0DAG5-9JsJ4A"; var docCod = SpreadsheetApp.openById(ID_PLANILHA_CODIGOS); var sheetCod = docCod.getSheets()[0]; var dadosCod = sheetCod.getRange(1, 1, sheetCod.getLastRow(), 2).getValues(); var listaCodigos = []; for (var k = 0; k < dadosCod.length; k++) { var numCod = String(dadosCod[k][0]).trim(); var descCod = String(dadosCod[k][1]).trim(); if (numCod !== "" && descCod !== "") { listaCodigos.push(numCod + " - " + descCod); } } return ContentService.createTextOutput(JSON.stringify({ "result": "success", "codigos": listaCodigos })).setMimeType(ContentService.MimeType.JSON); } // ------------------------------------------------------------------ // 6. SALVAMENTO UNIFICADO DA O.S // ------------------------------------------------------------------ var docOS = SpreadsheetApp.getActiveSpreadsheet(); var sheetOS = docOS.getSheetByName("Dados"); // LÊ A CÉLULA A1 DA PLANILHA PRINCIPAL NO MOMENTO DO ENVIO var numeroOSGerado = sheetOS.getRange("A1").getValue(); var isCross = (data.modelo_controlador === "Cross"); var newRowOS = [ "", numeroOSGerado, data.controladora || "", data.tipo_servico === "Corretivo" ? "X" : "", data.tipo_servico === "Preventivo" ? "X" : "", data.tipo_servico === "Implantação" ? "X" : "", data.data_inicial || "", data.hora_inicial || "", data.data_final || "", data.hora_final || "", data.descricao_servico || "", data.tensao_alternada || "", isCross ? (data.tensao_75vdc || "") : "", isCross ? (data.tensao_12vdc || "") : "", isCross ? (data.tensao_24vdc || "") : "", data.aterramento || "", data.cabeamento || "", data.fusivel || "", data.involucro || "", data.gabinetes || "", data.placas_cpu_potencia || "", data.padronizada || "", data.blocos_semaforicos || "", data.colunas || "", data.bracos || "", data.status_servico === "Concluído" ? "X" : "", data.status_servico === "Pendente" ? "X" : "", data.pendencia || "", data.material_texto || "", data.observacao || "", data.equipe || "", data.tecnico || "", data.supervisor || "", data.codigos_servico || "", data.nobreak || "", data.status_controlador === "Operacional" ? "X" : "", data.status_controlador === "Inoperante" ? "X" : "" ]; sheetOS.appendRow(newRowOS); // 6.2. SALVAMENTO NA PLANILHA DE ANÁLISE DE RISCO (MAPEAMENTO DAS COLUNAS A ATÉ I) if (data.trabalho_altura === "Sim") { try { var ID_PLANILHA_RISCO = "1S6G6v0TBOv0fCdjWoteCemhGDe5itL4RQz3QNQkn4i0"; var docRisco = SpreadsheetApp.openById(ID_PLANILHA_RISCO); var sheetRisco = docRisco.getSheets()[0]; // Estrutura das colunas: A: Carimbo | B: NSM | C: Local | D: Serviço | E: Data Inicial | F: Hora Inicial | G: Data Final | H: Hora Final | I: Trabalhadores sheetRisco.appendRow([ new Date(), numeroOSGerado, data.controladora || "", data.descricao_servico || "", data.data_inicial || "", data.hora_inicial || "", data.data_final || "", data.hora_final || "", data.trabalhadores_altura || "" ]); } catch (eRisco) { // Log secundário } } // 6.3. SALVAMENTO NA PLANILHA DE ALMOXARIFADO (ADICIONA LINHA VINCULADA À OS) if (data.usou_material && data.materiais_utilizados && data.materiais_utilizados.length > 0) { try { var ID_PLANILHA_ALMOX = "1RXvLOSMnhFRIZMvslt9npmZTbqZrLa9lBeOxL8o1lEA"; var docAlmox = SpreadsheetApp.openById(ID_PLANILHA_ALMOX); var sheetAlmox = docAlmox.getSheets()[0]; var cabecalhos = sheetAlmox.getRange(1, 1, 1, sheetAlmox.getLastColumn()).getValues()[0]; var linhaAlmox = new Array(cabecalhos.length).fill(""); // Coluna A (Índice 0): Carimbo de data/hora linhaAlmox[0] = new Date(); // Coluna B (Índice 1): Número da O.S linhaAlmox[1] = numeroOSGerado; // Preenche as quantidades nas colunas dos respectivos materiais data.materiais_utilizados.forEach(function(item) { for (var col = 2; col < cabecalhos.length; col++) { if (String(cabecalhos[col]).trim().toLowerCase() === String(item.nome).trim().toLowerCase()) { linhaAlmox[col] = item.qtd; break; } } }); sheetAlmox.appendRow(linhaAlmox); } catch (eAlmox) { // Log secundário } } return ContentService.createTextOutput(JSON.stringify({ "result": "success", "numero_os": numeroOSGerado })).setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ "result": "error", "error": error.toString() })) .setMimeType(ContentService.MimeType.JSON); } finally { lock.releaseLock(); } }
